cmake_minimum_required(VERSION 3.8)
project(linkerhand_description)
find_package(catkin REQUIRED)
catkin_package()
set(UPSTREAM_URDF "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor/linkerhand-urdf")
if(NOT EXISTS "${UPSTREAM_URDF}/L20lite/left/linkerhand_l20lite_left.urdf")
  message(FATAL_ERROR "Initialize the URDF submodule with git submodule update --init --recursive")
endif()
file(MAKE_DIRECTORY "${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_SHARE_DESTINATION}/models")
execute_process(COMMAND ${CMAKE_COMMAND} -E create_symlink
  "${UPSTREAM_URDF}/L20lite" "${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_SHARE_DESTINATION}/models/L20lite")
catkin_install_python(PROGRAMS scripts/load_urdf.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(PROGRAMS scripts/load_urdf.py DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/scripts)
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY "${UPSTREAM_URDF}/L20lite" DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/models)
